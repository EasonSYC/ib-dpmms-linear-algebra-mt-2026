\section{Matrices and Linear Maps}

Let \(V, W\) be linear maps. Define the set
\[
    \linearmap\pqty{V, W} = \set{\functiondomain{\alpha}{V}{W}}{\alpha \text{ is a linear map}}
\]
equipped with element-wise addition and scalar multiplication
\[
    \pqty{\alpha + \beta} \pqty{\vb{v}} = \alpha\pqty{\vb{v}} + \beta\pqty{\vb{v}} \qand \pqty{\lambda \alpha} \pqty{\vb{v}} = \lambda \alpha \pqty{\vb{v}}.
\]

It is straightforward to check that \(\linearmap\pqty{V, W}\) is an \(\FF\)-vector space.

Suppose onwards that \(V\) and \(W\) are finite-dimensional, with \(\dim \pqty{V} = n\) and \(\dim \pqty{W} = m\).

\subsection{Matrices and Bases}

\begin{theorem}[Linear Maps and Matrices]
    \label{thm:linear-maps-matrices}%
    We have
    \[
        \linearmap\pqty{V, W} \isomorphic \matrixset_{m \times n} \pqty{\FF}.
    \]

    In particular, we have \(\dim\pqty{\linearmap\pqty{V, W}} = mn\).
\end{theorem}

\begin{notation}
    For \(A \in \matrixset_{m \times n} \pqty{\FF}\), we denote the entries of \(A\) as \(A = \pqty{a_{i, j}}\) where \(a_{i, j} \in \FF\) and \(1 \leq i \leq m\), \(1 \leq j \leq n\). Furthermore, let \(\matrixset_{m \times n}\pqty{\FF}\) has a standard basis
    \[
        \set{E_{r, s}}{1 \leq r \leq m, 1 \leq s \leq n}
    \]
    where \(E_{r, s} = \pqty{\kronecker_{r, i}, \kronecker_{s, j}}\)
\end{notation}

Since \(V\) and \(W\) are finite dimensional, let \(\basis = \Bqty{\vb{v}_1, \ldots, \vb{v}_n}\) and \(\mathcal{C} = \Bqty{\vb{w}_1, \ldots, \vb{w}_m}\). Then given a linear map \(\alpha \in \linearmap\pqty{V, W}\), we may apply \(\alpha\) to the basis vectors \(\vb{v}_j\) to get linear combinations of \(\vb{w}_i\)
\begin{align*}
    \alpha\pqty{\vb{v}_1} &= a_{1, 1} \vb{w}_1 + a_{2, 1} \vb{w}_2 + \cdots + a_{m, 1} \vb{w}_m, \\
    \alpha\pqty{\vb{v}_2} &= a_{1, 2} \vb{w}_1 + a_{2, 2} \vb{w}_2 + \cdots + a_{m, 2} \vb{w}_m, \\
    \vdots &= \vdots \\
    \alpha\pqty{\vb{v}_n} &= a_{1, n} \vb{w}_1 + a_{2, n} \vb{w}_2 + \cdots + a_{m, n} \vb{w}_m.
\end{align*}

\begin{definition}[Matrix of Linear Map]
    The \term{matrix of the linear map} \(\alpha\) with respect to \(\basis\) and \(\mathcal{C}\) is given by
    \[
        \bqty{\alpha}_{\mathcal{C}}^{\basis} = \pqty{a_{i, j}} \in \matrixset_{m \times n} \pqty{\FF}
    \]
\end{definition}

The purpose of defining the matrix this way around is for consistent matrix multiplication with coordinate vectors. In particular, for all \(\vb{v} \in V\), we have
\[
    \bqty{\alpha}_{\mathcal{C}}^{\basis} \bqty{\vb{v}}_{\basis} = \bqty{\alpha\pqty{\vb{v}}}_{\mathcal{C}}.
\]

Furthermore, the matrix defined above is in fact the \emph{only} matrix \(A \in \matrixset_{m \times n} \pqty{\FF}\) that gives
\[
    A \bqty{\vb{v}}_{\basis} = \bqty{\alpha\pqty{\vb{v}}}_{\mathcal{C}}
\]
for all \(\vb{V} \in V\).

\begin{proof}
    \begin{enumerate}[label=(\roman*)]
        \item \textbf{Compatibility with Coordinate Vectors.} For any vector \(\vb{v} = \sum_{j = 1}^{n} \lambda_j \vb{v}_j\), we have
        \begin{align*}
            \alpha\pqty{\vb{v}} &= \sum_{j = 1}^{n} \lambda_j \alpha\pqty{\vb{v}_j} \\
            &= \sum_{j = 1}^{n} \lambda_j \sum_{i = 1}^{m} a_{ij} \vb{w}_i \\
            &= \sum_{i = 1}^{m} \pqty{\sum_{j = 1}^{n} a_{i, j} \lambda_j} w_i.
        \end{align*}
    
        Then note
        \[  
            \bqty{\alpha\pqty{\vb{v}}}_{\mathcal{C}} = \pqty{\sum_{j = 1}^{n} a_{ij} \lambda_j}_i = \pqty{a_{i, j}}_{i, j} \pqty{\lambda_j}_j = \bqty{\alpha}_{\mathcal{C}}^{\basis} \bqty{\vb{v}}_{\basis}
        \]
        is indeed the standard multiplication of the matrix and the basis coordinate vector.
    
        \item \textbf{Uniqueness with Coordinate Vectors}. By definition, \(\bqty{\vb{v_i}}_{\basis} = \vb{e}_i\) gives the standard basis vectors, so for \(A \in \matrixset_{m \times n} \pqty{\FF}\), we have
        \[
            A \bqty{\vb{v}_i}_{\mathcal{B}} = A \vb{e}_i = i\text{-th column of }A.
        \]
    
        But if another matrix \(A\) maps the corresponding coordinate vectors to the coordinate vectors, then
        \begin{align*}
            A \bqty{\vb{v}_i}_{\basis} &= \bqty{\alpha \pqty{\vb{v}_\mathcal{C}}}_{\mathcal{C}} \\
            &= \bqty{\alpha}_{\mathcal{C}}^{\basis} \bqty{\vb{v}_i}_{\basis} \\
            &= \bqty{\alpha}_{\mathcal{C}}^{\basis} \vb{e}_i \\
            &= i\text{-th colunn of } \bqty{\alpha}_{\mathcal{C}}^{\basis}
        \end{align*}
        and this holds for all \(i\), so \(A = \bqty{\alpha}_{\mathcal{C}}^{\basis}\).
    
        \item \textbf{Linearity.} For \(\alpha, \beta \in \linearmap\pqty{V, W}\) and \(\lambda \in \FF\), we can compare the matrices \(\bqty{\alpha}_{\mathcal{C}}^{\basis} + \lambda \bqty{\alpha}_{\mathcal{C}}^{\basis}\) with \(\bqty{\alpha + \lambda \beta}_{\mathcal{C}}^{\basis}\). For \(\vb{v} \in V\), we have
        \begin{align*}
            \pqty{\bqty{\alpha}_{\mathcal{C}}^{\basis} + \lambda \bqty{\beta}_{\mathcal{C}}^{\basis}} \bqty{\vb{v}}_{\basis} &= \bqty{\alpha \pqty{\vb{v}}}_{\mathcal{C}} + \lambda \bqty{\beta\pqty{\vb{v}}}_{\mathcal{C}} \\
            &= \bqty{\alpha \pqty{\vb{v}} + \lambda \beta \pqty{\vb{v}}}_{\mathcal{C}} \\
            &= \bqty{\pqty{\alpha + \lambda \beta}}_{\mathcal{C}}^{\basis} \bqty{\vb{v}}_{\basis}.
        \end{align*}
    
        Hence, by uniqueness in (ii), we have \(\bqty{\alpha + \lambda \beta}_{\mathcal{C}}^{\basis} = \bqty{\alpha}_{\mathcal{C}}^{\basis} + \lambda \bqty{\beta}_{\mathcal{C}}^{\basis}\).

        \item \textbf{Injectivity.} If \(\alpha, \beta \in \linearmap\pqty{V, W}\), with \(\bqty{\alpha}_{\mathcal{C}}^{\basis} = \bqty{\beta}_{\mathcal{C}}^{\basis}\), then we have
        \[  
            \alpha\pqty{\vb{v}_i} = \beta\pqty{\vb{v}_i}
        \]
        and by \autoref{prop:linear-map-is-determined-by-basis} (Uniqueness of Linear Extensions), we have \(\alpha = \beta\).

        \item \textbf{Surjectivity.} Given \(A \in \matrixset_{m \times n} \pqty{\FF}\), let \(\alpha in \linearmap\pqty{V, W}\) be the linear extension of
        \[
            \alpha\pqty{\vb{v}_j} = \sum_{i = 1}^{m} a_{i, j} \vb{w}_u
        \]
        with \(A = \pqty{a_{i, j}}\). Then \(\bqty{\alpha}_{\mathcal{C}}^{\basis} = A\).
    \end{enumerate}
\end{proof}

\begin{proposition}[Matrix of Composition]
    \label{prop:composition-matrix}%
    Let \(V, W, X\) be finite-dimensional vector spaces with bases \(\basis, \mathcal{C}, \mathcal{D}\) respectively, with \(\alpha \in \linearmap\pqty{V, W}\) and \(\beta \in \linearmap\pqty{W, X}\). Then
    \[
        \bqty{\beta \circ \alpha}_{\mathcal{D}}^{\basis} = \bqty{\beta}_{\mathcal{D}}^{\mathcal{C}} \bqty{\alpha}_{\mathcal{C}}^{\basis}.
    \]
\end{proposition}

\begin{proof}
    By \autoref{thm:linear-maps-matrices}, the matrix \(A = \bqty{\beta \circ \alpha}_{\mathcal{C}}^{\basis}\) is uniquely determined by
    \[  
        A \bqty{\vb{v}}_{\basis} = \bqty{\beta\pqty{\alpha\pqty{\vb{v}}}}_{\mathcal{D}}
    \]
    for all \(\vb{v} \in V\). But notice that
    \begin{align*}
        \pqty{\bqty{\beta}_{\mathcal{D}}^{\mathcal{C}} \bqty{\alpha}_{\mathcal{C}}^{\basis}} \bqty{\vb{v}}_{\basis} &= \bqty{\beta}_{\mathcal{D}}^{\mathcal{C}} \bqty{\alpha\pqty{\vb{v}}}_{\mathcal{C}} \\
        &= \bqty{\beta\pqty{\alpha\pqty{\vb{v}}}}_{\mathcal{D}}
    \end{align*}
    for all \(\vb{v} \in V\). Then we have \(A = \bqty{\beta}_{\basis}^{\mathcal{C}} \bqty{\alpha}_{\mathcal{C}}^{\basis}\).
\end{proof}

It is relatively obvious that for any basis \(\basis\) of \(V\), we have
\[
    \bqty{\identity_V}_{\basis}^{\basis} = I_n.
\]

\begin{definition}[Base-Change Matrix]
    Given bases \(\basis, \basis'\) of \(V\), the \term{base-change matrix} from \(\basis\) to \(\basis'\) is given by
    \[
        P = \bqty{\identity_V}_{\basis'}^{\basis} = \matrixset_{m \times n} \pqty{\FF}.
    \]

    Equivalently, for \(\basis = \Bqty{\vb{v}_1, \vb{v}_2, \ldots, \vb{v}_n}\) and \(\basis' = \Bqty{\vb{v}_1', \vb{v}_2', \ldots, \vb{v}_n'}\), then \(P = \pqty{p_{i, j}}\), where
    \[
        \vb{v}_j = \sum_{i = 1}^{n} p_{i, j} \vb{v}_i'
    \]
    so the \(j\)-th column of \(P\) is \(\bqty{\vb{v}_j}_{\basis'}\).
\end{definition}

Recall that a matrix \(A \in \matrixset_{n \times n} \pqty{\FF}\) is \term{invertible} if there exists \(A\inverse \in \matrixset_{m \times n} \pqty{\FF}\) such that \(A A\inverse = A\inverse A = I_n\), and we call \(A\inverse\) an \term{inverse} for \(A\). The \term{group} of invertible \(n \times n\) matrices over \(\FF\) is denoted \(\GL_n \pqty{\FF}\).

\begin{proposition}[Change of Basis Matrices]
    \label{prop:change-of-basis-matrix}%
    We have the following.

    \begin{enumerate}[label=(\roman*)]
        \item The matrix \(\bqty{\identity_V}_{\basis'}^{\basis}\) is invertible, with inverse \(\bqty{\identity_V}_{\basis}^{\basis'}\).
        \item \term{Change of basis formula.} For \(\alpha \in \linearmap\pqty{V, W}\) and bases \(\basis, \basis'\) for \(V\), \(\mathcal{C}, \mathcal{C}'\) for \(W\), we have
        \[
            \bqty{\alpha}_{\mathcal{C}'}^{\basis'} = \bqty{\identity_W}_{\mathcal{C}'}^{\mathcal{C}} \bqty{\alpha}_{\mathcal{C}}^{\basis} \bqty{\identity_V}_{\basis}^{\basis'}.
        \]
    \end{enumerate}
\end{proposition}

\begin{proof}
    Both are immediate from \autoref{prop:composition-matrix}.
\end{proof}

Note that the converse is true: any invertible matrix is a base-change matrix.

\begin{theorem}[Rank and Identity Submatrix]
    \label{thm:rank-and-identity-submatrix}%
    For any linear map \(\alpha \in \linearmap\pqty{V, W}\), we have the following.

    \begin{enumerate}
        \item There exists basis \(\basis, \mathcal{C}\) for \(V, W\) respectively such that
        \[
            \bqty{\alpha}_{\mathcal{C}}^{\basis} = \pmqty{I_r & 0 \\ 0 & 0}
        \]
        where \(r = \rank\pqty{\alpha}\).

        \item If there are some other bases \(\basis', \mathcal{C}'\) for \(V, W\) respectively such that
        \[
            \bqty{\alpha}_{\mathcal{C}'}^{\basis'} = \pmqty{I_{r'} & 0 \\ 0 & 0}
        \]
        then \(r = r'\).
    \end{enumerate}
\end{theorem}

\begin{proof}
    By \autoref{thm:rank-nullity-thm} (Rank--Nullity Theorem), \(\ker\pqty{\alpha}\) has a basis
    \[
        \Bqty{\vb{v}_{r + 1}, \ldots, \vb{v}_n},
    \]
    and we extend this to a basis for \(V\) by adding vectors \(\vb{v}_1, \ldots, \vb{v}_r\) to get
    \[
        \basis = \Bqty{\vb{v}_1, \ldots, \vb{v}_r, \vb{v}_{r + 1}, \ldots, \vb{v}_n}.
    \]
    
    Then \(\alpha\pqty{\vb{v}_1}, \ldots, \alpha\pqty{\vb{v}_r}\) spans \(\img\pqty{\alpha}\) with at most size \(r\). Then this is a basis for \(\img\pqty{\alpha}\), and denote
    \[
        \Bqty{\vb{w}_1 = \alpha\pqty{\vb{v}_1}, \ldots, \vb{w}_r = \alpha\pqty{\vb{v}_r}}
    \]
    as a basis of \(\img\pqty{\alpha}\). We extend this to a basis for \(W\) by adding vectors \(\vb{w}_{r + 1}, \ldots, \vb{w}_m\), to get
    \[
        \mathcal{C} = \Bqty{\vb{w}_1, \ldots, \vb{w}_r, \vb{w}_{r + 1}, \ldots, \vb{w}_m}.
    \]
    
    Then we have
    \[
        \alpha\pqty{\vb{v}_i} = \begin{cases}
            \vb{w}_i & 1 \leq i \leq r, \\
            \vb{0} & r + 1 \leq i \leq n.
        \end{cases}
    \]

    So indeed
    \[
        \bqty{\alpha}_{\mathcal{C}}^{\basis} = \pmqty{I_r & 0 \\ 0 & 0}.
    \]

    Conversely, if \(\basis', \mathcal{C}'\) are bases for \(V, W\) respectively such that
    \[
        \bqty{\alpha}_{\mathcal{C}'}^{\basis'} = \pmqty{I_{r'} & 0 \\ 0 & 0},
    \]
    then we have
    \[
        \alpha\pqty{\vb{v}_i'} = \begin{cases}
            \vb{w}_i' & 1 \leq i \leq r', \\
            \vb{0} & r' + 1 \leq i \leq n.
        \end{cases}
    \]

    Hence, \(\vb{w}_1', \ldots, \vb{w}_{r'}'\) spans \(\img \pqty{\alpha}\) and is linearly independent as a subset of the basis. So it is a basis for \(\img\pqty{\alpha}\), and we have \(r' = \rank\pqty{\alpha} = r\).
\end{proof}

\subsection{Applications of Matrices}

\begin{definition}[Equivalent Matrices]
    Let \(A, A' \in \matrixset_{m \times n} \pqty{\FF}\). We say that \(A\) and \(A'\) are \term{equivalent} if there exist invertible matrices \(P \in \GL_m\pqty{\FF}\) and \(Q \in \GL_n\pqty{\FF}\) such that
    \[
        A' = P A Q.
    \]
\end{definition}

\begin{remarks}
    \begin{itemize}
        \item This definition is motivated by the change of basis formula in \autoref{prop:change-of-basis-matrix}. In particular, if \(\bqty{\alpha}_{\mathcal{C}}^{\basis}\) and \(\bqty{\alpha}_{\mathcal{C}'}^{\basis'}\) are equivalent: matrices of the same linear map \(\alpha\) with respect to different bases, then they are equivalent.
        \item In addition, equivalence of matrices is an equivalence relation on \(\matrixset_{m \times n} \pqty{\FF}\), since \(\GL_m\pqty{\FF}\) and \(\GL_n\pqty{\FF}\) are groups under matrix multiplication.
    \end{itemize}
\end{remarks}

\begin{definition}[Column and Row Spaces]
    Given a matrix \(A \in \matrixset_{m \times n} \pqty{\FF}\), we define the \term{column space} of \(A\) as the subspace \(\Col\pqty{A} \subspace \FF^m\) spanned by the columns of \(A\), and the \term{column rank} of \(A\) is \(\colrank\pqty{A} = \dim\pqty{\Col\pqty{A}}\).
    
    The \term{row space} of \(A\) is the subspace \(\Row\pqty{A} = \Col\pqty{A \transpose} \subspace \FF^n\) spanned by the (transpose of) rows of \(A\), and the \term{row rank} of \(A\) is \(\rowrank\pqty{A} = \dim\pqty{\Row\pqty{A}}\).
\end{definition}

\begin{remark}
    For a matrix \(A \in \matrixset_{m \times n} \pqty{\FF}\), there is a linear map \(\functiondomain{\alpha}{\FF^n}{\FF^m}\) given by \(\alpha\pqty{\vb{x}} = A \vb{x}\). Then \(A = \bqty{\alpha}_{E_m}^{E_n}\) where \(E_m, E_n\) are the standard bases for \(\FF^m, \FF^n\) respectively. Moreover, notice \(\img\pqty{\alpha} = \Col\pqty{A}\), and \(\rank\pqty{\alpha} = \colrank\pqty{A}\).
\end{remark}

\begin{notation}
    We write \(\ker\pqty{A}, \img\pqty{A}, \rank\pqty{A}, \nul\pqty{A}\) for the kernel, image, rank and nullity of the linear map \(\alpha\) associated to \(A\).
\end{notation}

\begin{theorem}[Classificaiton of Matrices up to Equivalence]
    \label{thm:classification-of-matrices-up-to-equivalence}%
    Let \(A, A' \in \matrixset_{m \times n} \pqty{\FF}\). Then we have the following.
    \begin{enumerate}[label=(\roman*)]
        \item \(A\) is equivalent to the matrix \(\pmqty{I_r & 0 \\ 0 & 0}\) where \(r = \rank\pqty{A}\).
        \item \(A\) and \(A'\) are equivalent if and only if \(\colrank\pqty{A} = \colrank\pqty{A'}\).
    \end{enumerate}
\end{theorem}

We use the following lemma about the rank of a product of matrices.

\begin{lemma}[Rank of Product of Matrices]
    \label{lem:rank-of-product-of-matrices}%
    Let \(A \in \matrixset_{m \times n} \pqty{\FF}\) and \(B \in \matrixset_{n \times p} \pqty{\FF}\). Then \(\rank \pqty{AB} \leq \min\pqty{\rank\pqty{A}, \rank\pqty{B}}\).
\end{lemma}

\begin{proof}
    Clearly, we note \(\img \pqty{AB} \subspace \img\pqty{A}\), so \(\rank\pqty{AB} \leq \rank\pqty{A}\).

    On the other hand, if \(B \vb{v} = \vb{0}\), then \(AB \vb{v} = \vb{0}\). Hence, \(\ker\pqty{B} \subspace \ker\pqty{AB}\), and by \autoref{thm:rank-nullity-thm} (Rank--Nullity Theorem), we have
    \[
        \rank\pqty{AB} = p - \nul\pqty{AB} \leq p - \nul\pqty{B} = \rank\pqty{B}.
    \]
    so \(\rank \pqty{AB} \leq \rank \pqty{B}\).
\end{proof}

Now we proof \autoref{thm:classification-of-matrices-up-to-equivalence}.

\begin{proof}[\proofname\ of \autoref{thm:classification-of-matrices-up-to-equivalence}]
    Let \(\alpha\) be as the linear map associated to \(A\), so \(A = \bqty{\alpha}_{E_m}^{E_n}\). So by \autoref{thm:rank-and-identity-submatrix}, there exists bases \(\basis, \mathcal{C}\) for \(\FF^n, \FF^m\) respectively such that
    \[
        \bqty{\alpha}_{\mathcal{C}}^{\basis} = \pmqty{I_r & 0 \\ 0 & 0}
    \]
    where \(r = \rank\pqty{\alpha} = \colrank\pqty{A}\). By \autoref{prop:change-of-basis-matrix}, we have
    \[
        \bqty{\alpha}_{\mathcal{C}}^{\basis} = \bqty{\identity_{\FF^m}}_{\mathcal{C}}^{E_m} \bqty{\alpha}_{E_m}^{E_n} \bqty{\identity_{\FF^n}}_{E_n}^{\basis} = P A Q
    \]
    where \(P = \bqty{\identity_{\FF^m}}_{\mathcal{C}}^{E_m} \in \GL_m\pqty{\FF}\) and \(Q = \bqty{\identity_{\FF^n}}_{E_n}^{\basis} \in \GL_n\pqty{\FF}\). So indeed \(A\) is equivalent to \(\pmqty{I_r & 0 \\ 0 & 0}\).

    For the second part, if \(A\) and \(A'\) have a common column rank, then \(A\) and \(A'\) are both equivalent to \(\pmqty{I_r & 0 \\ 0 & 0}\) where \(r = \colrank\pqty{A} = \colrank\pqty{A'}\). Hence, \(A\) and \(A'\) are equivalent.

    If \(A\) and \(A'\) are equivalent, say \(A' = P A Q\) for some \(P \in \GL_m\pqty{\FF}\) and \(Q \in \GL_n\pqty{\FF}\), then by \autoref{lem:rank-of-product-of-matrices}, we have
    \[
        \rank\pqty{A'} = \rank\pqty{P \pqty{AQ}} \leq \rank\pqty{AQ} \leq \rank\pqty{A}.
    \]
    and symmetrically, \(\rank\pqty{A} \leq \rank\pqty{A'}\). Hence, \(\colrank\pqty{A} = \colrank\pqty{A'}\).
\end{proof}

\begin{theorem}[Row Rank is equal to Column Rank]
    For any matrix \(A \in \matrixset_{m \times n} \pqty{\FF}\), we have
    \[
        \colrank\pqty{A} = \rowrank\pqty{A}.
    \]
\end{theorem}

\begin{proof}
    By \autoref{thm:classification-of-matrices-up-to-equivalence}, we may write
    \[
        PAQ = \pmqty{I_r & 0 \\ 0 & 0} \in \matrixset_{m \times n} \pqty{\FF}
    \]
    where \(r = \colrank\pqty{A}\). Transposing both sides, we have
    \[
        Q\transpose A\transpose P\transpose = \pmqty{I_r & 0 \\ 0 & 0} \in \matrixset_{n \times m} \pqty{\FF},
    \]
    and in particular, if \(P\) and \(Q\) are invertible, then \(P\transpose\) and \(Q\transpose\) are also invertible. [Note this does \emph{not} mean that \(PAQ = Q\transpose A\transpose P\transpose\) since the matrices are of different sizes.] Hence, by \autoref{thm:classification-of-matrices-up-to-equivalence}, we have
    \[
        \rowrank\pqty{A} = \colrank\pqty{A\transpose} = r = \colrank\pqty{A}
    \]
    finishing the proof.
\end{proof}

Now we reduce to consider only square matrices.

\begin{remark}
    Let \(\basis, \basis'\) be bases for \(V\), and let \(\alpha \in \linearmap\pqty{V, V}\) be an endomorphism. Then by \autoref{prop:change-of-basis-matrix}, we have
    \[
        \bqty{\alpha}_{\basis'}^{\basis'} = \bqty{\identity_V}_{\basis'}^{\basis} \bqty{\alpha}_{\basis}^{\basis} \bqty{\identity_V}_{\basis}^{\basis'},
    \]
    and if \(P = \bqty{\identity_V}_{\basis'}^{\basis}\), then \(P\) is invertible with \(P\inverse = \bqty{\identity_V}_{\basis}^{\basis'}\).
\end{remark}

The following definition is motivated by the above remark.

\begin{definition}[Similar Matrices]
    For \(A, A' \in \matrixset_{n \times n} \pqty{\FF}\), we say that \(A\) and \(A'\) are \term{similar} if there exists an invertible matrix \(P \in \GL_n\pqty{\FF}\) such that
    \[
        A' = P A P\inverse.
    \]
\end{definition}

\begin{remarks}
    \begin{itemize}
        \item Similarity of matrices is an equivalence relation on \(\matrixset_{n \times n} \pqty{\FF}\).
        \item If \(A, A'\) are similar, then they are equivalent, but the converse is not true.
    \end{itemize}
\end{remarks}

\begin{example}[Equivalent Matrices that are not Similar]
    Note that \(I_n \in \matrixset_{n \times n} \pqty{\FF}\) is equivalent to every \(P \in \GL_n\pqty{\FF}\), but \(I_n\) is only similar to itself.
\end{example}

Determining the equivalence classes of matrices up to similarity is a much more difficult problem than determining the equivalence classes of matrices up to equivalence. In a sense, this is what the majority of the remaining of this course is about, relating to the eigenvalues and eigenvectors, and the Jordan canonical form of a matrix.
