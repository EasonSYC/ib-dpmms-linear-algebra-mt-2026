\section{Span, Linear Independence, Basis, and Dimension}

\subsection{Spanning Set and Linear Independence}

\begin{definition}[Span of a Set, Spanning Set]
    Let \(S \subseteq V\). The \term{span} of \(S\) in \(V\) is given by
    \[
        \spn \pqty{S} = \angBrac{S} = \set{{\sum_{\vb{s} \in S}}' \lambda_{\vb{s}} \vb{s}}{\lambda_{\vb{s}} \in \FF},
    \]
    where the sum \({\sum}'\) denotes a \term{finite sum}, i.e., we require that \(\lambda_{\vb{s}} = 0\) for all but finitely many \(\vb{s} \in S\).
    
    The elements of \(\spn\pqty{S}\), in the finite sum form, are called \term{linear combinations} of elements of \(S\).

    If \(\spn\pqty{S} = V\), then we say \(S\) is a \term{spanning set} of \(V\).
\end{definition}

\begin{remark}
    Note that we require \({\sum}'\), since there is no well-defined way to sum infinitely many vectors in a vector space (this requires \emph{analysis}). In particular, if \(S\) is infinite, then \(\spn\pqty{S}\) is the set of all finite linear combinations of elements of \(S\).
\end{remark}

\begin{definition}[Finite/Infinite-Dimensional Vector Space]
    We say that a vector space \(V\) is \term{finite-dimensional} if there exists a finite spanning set \(S \subseteq V\). Otherwise, we say that \(V\) is \term{infinite-dimensional}.
\end{definition}

\begin{remarks}
    \begin{itemize}
        \item \(\spn\pqty{S} \subspace V\) for any \(S \subseteq V\).
        \item By convention, \(\spn\pqty{\emptyset} = \Bqty{\vb{0}_V}\).
        \item If \(S\) spans \(V\) and \(S \subseteq \spn\pqty{T}\), then \(T\) spans \(V\).
        \item \(\spn\pqty{S \cup T} = \spn\pqty{S} + \spn\pqty{T}\).
    \end{itemize}
\end{remarks}

\begin{examples}
    \begin{enumerate}[label=(\roman*)]
        \item Let \(V = \RR^3\), and let
        \[
            S = \Bqty{\pmqty{1 \\ 0 \\ 0}, \pmqty{1 \\ 1 \\ 2}} \qand T = \Bqty{\pmqty{2 \\ 1 \\ 2}, \pmqty{0 \\ 1 \\ 2}, \pmqty{-1 \\ 2 \\ 4}}.
        \]

        Then
        \[
            \spn\pqty{S} = \spn\pqty{T} = \set{\pmqty{x \\ y \\ 2y}}{x, y \in \RR} \subspace \RR^3.
        \]

        \item Let \(V = \RR^3\), and let
        \[
            T = \set{\kronecker_n}{n \in \NN} \subseteq V
        \]
        where \(\functiondomain{\kronecker_n}{\NN}{\RR}\) is given by
        \[
            \kronecker_n\pqty{m} = \kronecker_{n, m} = \begin{cases}
                1 & n = m, \\
                0 & n \neq m.
            \end{cases}
        \]

        Then \(\spn\pqty{T} \neq \RR^\NN\). For instance, if \(f \in \RR^\NN\) satisfies \(f\pqty{n} = 1\) for all \(n\), then \(f \notin \spn\pqty{T}\). In fact,
        \[
            \spn\pqty{T} = \set{g \in V}{g\pqty{n} = 0 \text{ for sufficiently large }n \in \NN}.
        \]
    \end{enumerate}
\end{examples}

\begin{definition}[Linearly Independence/Dependence]
    A set \(S \subseteq V\) is \term{linearly independent (L.I.)} if, for any linear combination satisfying
    \[
        {\sum_{\vb{s} \in S}}' \lambda_{\vb{s}} \vb{s} = \vb{0}_V,
    \]
    we have \(\lambda_{\vb{s}} = 0\) for all \(\vb{s} \in S\). Otherwise, we say that \(S\) is \term{linearly dependent (L.D.)}.
\end{definition}

\begin{remarks}
    \begin{itemize}
        \item If a set \(S \subseteq V\) is linearly independent, then \(\vb{0} \notin S\).
        \item If \(S\) is finite, with \(S = \Bqty{\vb{v}_1, \vb{v}_2, \ldots, \vb{v}_n}\) and \(\abs{S} = n\), then \(S\) is linearly independent, if and only if for all \(\lambda_1, \lambda_2, \ldots, \lambda_n \in \FF\), that
        \[
            \sum_{i = 1}^{n} \lambda_i \vb{v}_i = \vb{0} \implies \lambda_i = \cdots = \lambda_n = 0.
        \]
        \item If \(S\) is infinite, then \(S\) is linearly independent, if and only if every finite subset of \(S\) is linearly independent.
        \item Every subset of a linearly independent set is itself linearly independent.
    \end{itemize}
\end{remarks}

\begin{examples}
    \begin{enumerate}[label=(\roman*)]
        \item Let \(V = \RR^3\), and let
        \[
            S = \Bqty{\pmqty{1 \\ 0 \\ 0}, \pmqty{1 \\ 1 \\ 2}} \qand T = \Bqty{\pmqty{2 \\ 1 \\ 2}, \pmqty{0 \\ 1 \\ 2}, \pmqty{-1 \\ 2 \\ 4}}.
        \]

        Then \(S\) is linearly independent, while \(T\) is linearly dependent, since
        \[
            1 \cdot \pmqty{2 \\ 1 \\ 2} - 5 \cdot \pmqty{0 \\ 1 \\ 2} + 2 \cdot \pmqty{-1 \\ 2 \\ 4} = \pmqty{0 \\ 0 \\ 0}.
        \]

        On the other hand, if \(\lambda, \mu \in \RR\) is such that
        \[
            \vb{0} = \lambda \pmqty{1 \\ 0 \\ 0} + \mu \pmqty{1 \\ 1 \\ 2} = \pmqty{\lambda + \mu \\ \mu \\ 2\mu}
        \]
        then \(\lambda = \mu = 0\), so \(S\) is linearly independent.

        \item Let \(V = \RR^3\), and let
        \[
            T = \set{\kronecker_n}{n \in \NN} \subseteq V
        \]
        where \(\functiondomain{\kronecker_n}{\NN}{\RR}\) is given by
        \[
            \kronecker_n\pqty{m} = \kronecker_{n, m} = \begin{cases}
                1 & n = m, \\
                0 & n \neq m.
            \end{cases}
        \]

        Let \(f = \sum_{n \in S \subset \NN}\lambda_n \delta_n\) where \(\lambda_n \in \RR\) and for some  finite set \(S \subset \NN\). If \(f = 0\), then for all \(m \in \NN\),
        \[
            0 = f\pqty{m} = \pqty{{\sum_{n \in S} \lambda_n \delta_n}} \pqty{m} = \sum_{n \in S} \lambda_n \delta_n\pqty{m} = \lambda_m,
        \]
        precisely giving the trivial linear combination. so \(T\) is linearly independent.
    \end{enumerate}
\end{examples}

\subsection{Basis}

\begin{definition}[Basis]
    Let \(S \subseteq V\). \(S\) is a \term{basis} of \(V\) if it is linearly independent and spans \(V\).
\end{definition}

\begin{examples}
    \begin{enumerate}
        \item Let \(\vb{e}_i \in \FF^n\) be the \term{\(i\)-th standard basis vector}, with entries 0 except for the \(i\)-th entry, which is 1. Then the set \(\Bqty{\vb{e}_i}{1 \leq i \leq n}\) is the \term{standard basis} of \(\FF^n\).
        \item The set
        \[
            \Bqty{\pmqty{1 \\ 0 \\ 0}, \pmqty{1 \\ 1 \\ 0}, \pmqty{1 \\ 1 \\ 1}} \subseteq \FF^3
        \]
        is a basis for \(\FF^3\).
        \item Let \(p_n \in P\pqty{\RR}\) be given by \(p_n\pqty{x} = x^n\). Then the set \(\Bqty{p_n}{n \in \NN}\) is a basis for \(P\pqty{\RR}\).
    \end{enumerate}
\end{examples}

\begin{proposition}[Finite Spanning Set reduces to a Basis]
    Suppose \(S \subseteq V\) is a finite spanning set. Then there exists a subset \(S' \subseteq S\) such that \(S\) is a basis of \(V\).
\end{proposition}

\begin{proof}
    If \(S\) is linearly independent, then we are done. If not, write
    \[
        S = \Bqty{\vb{v}_1, \vb{v}_2, \ldots, \vb{v}_n}.
    \]

    There exists \(\lambda_1, \lambda_2, \ldots, \lambda_n \in \FF\), not all zero, such that
    \[
        \sum_{i = 1}^{n} \lambda_i \vb{v}_i = \vb{0}.
    \]

    Without loss of generality, up to relabelling, that \(\lambda_1 \neq 0\). Then rearranging this expression,
    \[
        \vb{v}_1 = - \frac{1}{\lambda_1} \sum_{i = 2}^{n} \lambda_i \vb{v}_i \in \spn\pqty{S \setminus \Bqty{\vb{v}_1}}.
    \]

    So \(S \setminus \Bqty{\vb{v}_1}\) also spans \(V\). Repeat this and this will terminate after at most \(n\) steps.
\end{proof}

\begin{corollary}[Finite-Dimensional Vector Space has a Finite Basis]
    \label{cor:finite-dimensional-vector-space-has-finite-basis}%
    Every finite-dimensional vector space has a finite basis.
\end{corollary}

\begin{remark}
    We will \emph{not} prove that every vector space has a basis. Some non-finite-dimensional vector spaces do have a basis, for example, the above set of polynomial functions over \(\RR\). However, even attempting to write down a basis of \({\RR}^{\NN}\) will be a lot of trouble -- in fact you cannot write down a basis of this space explicitly. The existence of a basis for every vector space is equivalent to the \emph{axiom of choice}, which is a controversial axiom in set theory. In other words, one should not assume that a infinite-dimensional vector space has a basis, unless it is explicitly constructed.
\end{remark}

\begin{theorem}[Steinitz Exchange Lemma]
    \label{thm:steinitz-exchange-lemma}%
    Let \(V\) be a \(\FF\)-vector space, \(S, T \subseteq V\) finite, with \(S\) being linearly-independent, and \(T\) spanning \(V\). Then we have the following.

    \begin{itemize}
        \item \emph{Linear independence set is always smaller than spanning set.} \(\abs{S} \leq \abs{T}\).
        \item \emph{Removing elements from a spanning set and replacing with linearly independent elements.} There exists \(T' \subseteq T\), such that \(S \cup T'\) spans \(V\), and \(\abs{T'} = \abs{T} - \abs{S}\).
    \end{itemize}
\end{theorem}

\begin{proof}
    Let \(S = \Bqty{\vb{s}_1, \ldots, \vb{s}_m}\) nad \(T = \Bqty{\vb{t}_1, \ldots, \vb{t}_n}\), with \(\abs{S} = m\) and \(\abs{T} = n\).

    If \(S = \emptyset\), then we are done. If not, we conduct the following steps.

    \textbf{Step 1.} Since \(\spn\pqty{T} = V\), then there exists \(\lambda_i \in \FF\), such that
    \[
        \vb{s}_1 = \sum_{i = 1}^{N} \lambda_i \vb{t}_i,
    \]
    and since \(\vb{s}_1 \neq \vb{0}\), so \(\lambda_i\) cannot be trivially all zeros. Without loss of generality, up to relabelling, we can assume that \(\lambda_1 \neq 0\). Then rearranging this expression,
    \[
        \vb{t}_1 = \frac{1}{\lambda_1} \pqty{\vb{s}_1 - \sum_{i = 2}^{N} \lambda_i \vb{t}_i} \in \spn\pqty{\Bqty{\vb{s}} \cup \Bqty{\vb{t}_2, \ldots, \vb{t}_n}}.
    \]

    \textbf{Step \(\boldsymbol{\pqty{k + 1}}\)} where \(k < m\). Assume the set \(\Bqty{\vb{s}_1, \ldots, \vb{s}_k, \vb{t}_{k + 1}}, \ldots, \vb{t}_n\) spans \(V\). Write
    \[
        \vb{s}_{k + 1} = \sum_{i = k + 1}^{n} \lambda_i \vb{t}_i + \sum_{j = 1}^{k} \mu_j \vb{s}_j,
    \]
    where \(\lambda_i, \mu_j \in \FF\), not all \(0\).

    If all \(\lambda_i = 0\), then
    \[
        \vb{s}_{k + 1} = \sum_{j = 1}^{k} \mu_j \vb{s}_j = \vb{0}
    \]
    but \(S\) is linearly independence, giving a contradiction.

    Hence, without loss of generality, \(\lambda_{k + 1} \neq 0\), so we can write
    \[
        \vb{t}_{k + 1} = \frac{1}{\lambda_{k + 1}} \pqty{\vb{s}_{k + 1} - \sum_{i = k + 2}^{n} \lambda_i \vb{t}_i - \sum_{j = 1}^{k} \mu_j \vb{s}_j} \in \spn\pqty{\Bqty{\vb{s}_1, \ldots, \vb{s}_{k + 1}} \cup \Bqty{\vb{t}_{k + 2}, \ldots, \vb{t}_n}}.
    \]

    So the new set \(\Bqty{\vb{s}_1, \ldots, \vb{s}_{k + 1}, \vb{t}_{k + 2}, \ldots, \vb{t}_n}\) spans \(V\).

    We repeat the process until either we run out of elements of \(S\), or elements of \(T\). But if we exhausted \(T\) before \(S\), then we would have a linear combination of elements of \(S\) that gives \(\vb{0}\), contradicting the linear independence of \(S\). Hence, we must have exhausted \(S\) first, and we have \(\abs{S} \leq \abs{T}\), which gives \(m \leq n\), and
    \[
        \Bqty{\vb{s}_1, \ldots, \vb{s}_m, \vb{t}_{m + 1}, \ldots, \vb{t}_n} \subseteq V
    \]
    with \(T' = \Bqty{\vb{t}_{m + 1}, \ldots, \vb{t}_n}\).
\end{proof}

\subsection{Dimension}

\begin{corollary}[Basis of Finite-Dimensional Vector Space]
    \label{cor:basis-of-finite-dimensional-vector-space}%
    Let \(V\) be a finite-dimensional vector space. Then every basis for \(V\) is finite, and any two bases \(\basis_1, \basis_2\) of \(V\) satisfy \(\abs{\basis_1} = \abs{\basis_2}\).
\end{corollary}

\begin{proof}
    By \autoref{cor:finite-dimensional-vector-space-has-finite-basis}, \(V\) has some finite basis \(\basis\). Suppose \(\basis'\) is an infinite basis.

    Let \(\basis'' \subset \basis'\) with \(\abs{\basis''} = \abs{\basis} + 1\). Then \(\basis''\) is linearly-independent, and \(\basis\) spans \(V\), contradicting with the first part of \autoref{thm:steinitz-exchange-lemma}.

    If \(\basis_1, \basis_2\) are bases of \(V\), then both must be finite. By the first part of \autoref{thm:steinitz-exchange-lemma}, we have \(\abs{\basis_1} \leq \abs{\basis_2}\) and \(\abs{\basis_2} \leq \abs{\basis_1}\), so \(\abs{\basis_1} = \abs{\basis_2}\).
\end{proof}

\begin{definition}[Dimension]
    For a finite-dimensional vector space \(V\), its \term{dimension} \(\dim\pqty{V}\) is the common size of any basis of \(V\).
\end{definition}

\begin{example}
    We have \(\dim\pqty{\FF^n} = n\).
\end{example}

\begin{corollary}[Size of Linearly Independent Set and Spanning Set]
    \label{cor:size-of-li-and-spanning}%
    Let \(V\) be a finite-dimensional vector space. Let \(S\) be a linearly independent set of \(V\), and \(T\) be a spanning set of \(V\). Then
    \[
        \abs{S} \leq \dim V \leq \abs{T},
    \]
    with equality on the left if and only if \(S\) is a basis, and equality on the right if and only if \(T\) is a basis.
\end{corollary}

\begin{proof}
    The inequalities themselves follow immediately from \autoref{thm:steinitz-exchange-lemma} (Steinitz).

    If \(S\) is a basis, then \(\abs{S} = \dim V\) by \autoref{cor:basis-of-finite-dimensional-vector-space}. Conversely, if \(\abs{S} = \dim V\), let \(\basis\) be a basis, then applying \autoref{thm:steinitz-exchange-lemma} (Steinitz) to \(S\) and \(\basis\), we have some \(\basis' \subseteq \basis\) such that \(S \cup \basis'\) spans \(V\), and \(\abs{\basis'} = \abs{\basis} - \abs{S} = 0\). Hence, \(S\) spans \(V\), and is a basis. The argument for \(T\) is similar.
\end{proof}

\begin{proposition}[Dimension of Subspace]
    If \(U \subspace V\) is a subspace of a finite-dimensional vector space \(V\), then \(U\) is also finite-dimensional, and \(\dim U \leq \dim V\), with equality if and only if \(U = V\).
\end{proposition}

\begin{proof}
    If \(U = \Bqty{\vb{0}}\), then \(\dim U = 0\), and we are done. Then, we construct a finite basis for \(U\) as follows.

    \textbf{Step 1.} Let \(\vb{0} \neq \vb{u}_1 \in U\). Then \(\Bqty{\vb{u}_i}\) is linearly independent.

    After \textbf{Step \(\boldsymbol k\).} We have \(S_k = \Bqty{\vb{u}_1, \ldots, \vb{u}_k} \subset U\) is linearly independent.

    By \autoref{cor:size-of-li-and-spanning}, we have \(k \leq \dim \pqty{V}\). If \(\spn S_k = U\), then we are done. If not, then there exists some \(\vb{u}_{k + 1} \in U \setminus \spn S_k\), and set
    \[
        S_{k + 1} = S_k \cup \Bqty{\vb{u}_{k + 1}} = \Bqty{\vb{u}_1, \ldots, \vb{u}_{k + 1}}.
    \]

    We claim that \(S_{k + 1}\) is linearly independent. Suppose not, then there exists \(\lambda_1, \ldots, \lambda_{k + 1} \in \FF\), not all zero, such that
    \[
        \sum_{i = 1}^{k + 1} \lambda_i \vb{u}_i = \vb{0}.
    \]

    If \(\lambda_{k + 1} = 0\), then we have a non-trivial linear combination of \(S_k\) that gives \(\vb{0}\), contradicting the linear independence of \(S_k\). Hence, \(\lambda_{k + 1} \neq 0\), and rearranging this expression, we can express \(\vb{u}_{k + 1}\) in terms of the elements of \(S_k\), contradicting the fact that \(\vb{u}_{k + 1} \notin \spn S_k\).

    This terminates in \(\leq \dim\pqty{V}\) steps to form a basis for \(U\), so \(U\) is finite-dimensional, and \(\dim U \leq \dim V\). Furthermore, if \(\dim \pqty{U} = \dim \pqty{V}\), using \autoref{cor:size-of-li-and-spanning} tells us that the basis of \(U\) is also a basis of \(V\), so \(U = V\).
\end{proof}

\begin{proposition}[Extending a Basis]
    Let \(U \subspace V\). For \(\basis_U\) a basis for \(U\), there exists a basis \(\basis_V\) for \(V\) such that \(\basis_U \subseteq \basis_V\).
\end{proposition}

\begin{proof}
    Applying the second part of \autoref{thm:steinitz-exchange-lemma} (Steinitz) with \(S = \basis_U\) and \(T\) for \emph{any} basis over \(V\), we obtain some \(T' \subseteq T\) such that \(\basis_V = \basis_U \cup T'\) spans \(V\), and \(\abs{T'} = \abs{T} - \abs{S} = \dim V - \dim U\).

    Then \(\abs{\basis_V} \leq \abs{\basis_U} + \abs{T'} = \dim V\), hence \(\basis_V\) is a basis for \(V\), by \autoref{cor:size-of-li-and-spanning}.
\end{proof}

\subsection{Basis and Linear Maps}

\begin{definition}[Rank and Nullity]
    We define the \term{rank} of a linear map \(\alpha\) as
    \[
        \rank \alpha = \dim \img \alpha
    \]
    and the \term{nullity} of a linear map \(\alpha\) as
    \[
        \nul \alpha = \dim \ker \alpha.
    \]
\end{definition}

\begin{theorem}[Rank--Nullity Theorem]
    Let \(\functiondomain{\alpha}{V}{W}\) be a linear map and let \(V\) be finite-dimensional. Then \(\img \alpha\) is finite-dimensional, and
    \[
        \dim \ker \alpha + \dim \img \alpha = \dim V.
    \]
\end{theorem}

\begin{proof}
    Let \(\Bqty{\vb{v}_1, \ldots, \vb{v}_k}\) be a basis for \(\ker \alpha\), and we extend this to a basis for \(V\) by adding \(\Bqty{\vb{v}_{k + 1}, \ldots, \vb{v}_n}\).

    We claim that the set
    \[
        \Bqty{\alpha\pqty{\vb{v}_{k + 1}}, \ldots, \alpha\pqty{\vb{v}_n}}
    \]
    forms a basis for \(\img \alpha\).

    \textbf{Spanning.} Let \(\vb{w} \in \img \alpha\) and write \(\vb{w} = \alpha\pqty{\vb{v}}\) for \(\vb{v} \in V\). Then we write
    \[
        \vb{v} = \sum_{i = 1}^{n} \lambda_i \vb{v}_i
    \]
    for some \(\lambda_i \in \FF\). Then by the linearity of \(\alpha\),
    \begin{align*}
        \vb{w} &= \alpha\pqty{\vb{v}} \\
        &= \sum_{i = 1}^{n} \lambda_i \alpha\pqty{\vb{v}_i} \\
        &= \sum_{i = k + 1}^{n} \lambda_i \alpha\pqty{\vb{v}_i},
    \end{align*}
    showing that the set is indeed a spanning set.

    \textbf{Linear independence.} Suppose that
    \[
        \vb{0} = \sum_{i = k + 1}^{l} \lambda_i \alpha\pqty{\vb{v}_i} = \alpha \pqty{\sum_{i = k + 1}^{n} \lambda_i \vb{v}_i}.
    \]

    Then
    \[
        \sum_{i = k + 1}^{n} \lambda_i \vb{v}_i = - \sum_{i = 1}^{k} \lambda_i \vb{v}_i
    \]
    for some \(\lambda_1, \ldots, \lambda_k \in \FF\). But this gives that
    \[
        \sum_{i = 1}^{n} \lambda_i \vb{v}_i = \vb{0},
    \]
    so all \(\lambda_i = 0\), and in particular, \(\lambda_{k + 1} = \cdots = \lambda_n = 0\), giving the trivial linear combination.
\end{proof}

\begin{corollary}[Dimension of Quotient Space]
    \label{cor:dimension-of-quotient-space}%
    Let \(U \subspace V\) be finite-dimensional. Then \(\flatfrac{V}{U}\) is finite-dimensional, and
    \[
        \dim \pqty{\flatfrac{V}{U}} = \dim V - \dim U.
    \]
\end{corollary}

\begin{proof}
    Recall that there is a surjective linear map
    \[
        \function{\pi_U}{V}{\flatfrac{V}{U}}{v}{v + U}
    \]
    which has \(\ker \pqty{\pi_U} = U\). Applying rank--nullity theorem, this gives the result as desired.
\end{proof}

\begin{remark}
    We could prove \autoref{cor:dimension-of-quotient-space} directly, by taking a basis of \(U\) and extending it to a basis of \(V\), and then argue as in our proof of the rank--nullity theorem. In particular, this approach gives an explicit basis for the quotient base: given basis \(\basis_U\) for \(U\) and \(\basis_V\) for \(V\) with \(\basis_U \subseteq \basis_V\), then the set
    \[
        \set{\vb{b} + U}{\vb{b} \in \basis_V \setminus \basis_U}
    \]
    forms a basis for the quotient space \(\flatfrac{V}{U}\). With this, we may simply deduce the result by \autoref{thm:first-isomorphism-theorem} (First Isomorphism Theorem).
\end{remark}

\begin{corollary}[Linear Map between Vector Spaces of Same Dimensions]
    If \(V\) and \(W\) are finite-dimensional with \(\dim V = \dim W\), and \(\functiondomain{\alpha}{V}{W}\) be a linear map. Then the following are equivalent.
    \begin{enumerate}[label=(\roman*)]
        \item \(\alpha\) is injective.
        \item \(\alpha\) is surjective.
        \item \(\alpha\) is an isomorphism.
    \end{enumerate}
\end{corollary}

\begin{proof}
    Recall that \(\alpha\) is injective if and only if \(\ker \alpha = \Bqty{\vb{0}}\), which is equivalent to \(\nul \alpha = 0\). Similarly, recall \(\alpha\) is surjective if and only if \(\img \alpha = W\), which is equivalent to \(\rank \alpha = \dim W = \dim V\).

    By rank--nullity theorem, we have
    \[
        \nul \alpha + \rank \alpha = \dim V.
    \]
    which gives the equivalence of the three statements.
\end{proof}

\begin{proposition}[Linear Map is Exactly Determined by its Action on a Basis]
    Let \(V, W\) be \(\FF\)-vector spaces (not necessarily finitely dimensional). Suppose \(\basis\) is a basis for \(V\).

    For any function \(\functiondomain{f}{\basis}{W}\), there is a unique linear map \(\functiondomain{F}{V}{W}\) extending \(f\), i.e., \(\rightbar{F}_{\basis} = f\).
\end{proposition}

\begin{proof}
    \textbf{Existence.} Given \(\vb{v} \in V\), write
    \[
        \vb{v} = {\sum_{\vb{b} \in \basis}}' \lambda_{\vb{b}} \vb{b}
    \]

    Then, we define
    \[
        F\pqty{\vb{v}} = {\sum_{\vb{b} \in \basis}}' \lambda_{\vb{b}} f\pqty{\vb{b}}.
    \]

    \textbf{Well-defined.} If \(\vb{v} = {\sum_{\vb{b} \in \basis}}' \mu_i \vb{b}\), then \(\vb{0} = \Bqty{\sum_{\vb{b} \in \basis}}' \pqty{\lambda_{\vb{b}} - \mu_{\vb{b}}} \vb{b}\), hence \(\lambda_{\vb{b}} = \mu_{\vb{b}}\) for all \(\vb{b} \in \basis\), and so \(F\) is well-defined.

    \textbf{Uniqueness.} Suppose \(G\) is a linear map extending \(f\). Then for any \(\vb{v} \in V\), we have
    \begin{align*}
        G\pqty{\vb{v}} &= G\pqty{{\sum_{\vb{b} \in \basis}}' \lambda_{\vb{b}} \vb{b}} \\
        &= {\sum_{\vb{b} \in \basis}}' \lambda_{\vb{b}} G\pqty{\vb{b}} \\
        &= {\sum_{\vb{b} \in \basis}}' \lambda_{\vb{b}} f\pqty{\vb{b}} \\
        &= F\pqty{\vb{v}}.
    \end{align*}

    \textbf{Linearity.} For \(\vb{u}, \vb{v} \in V\) and \(\lambda \in \FF\), let
    \[
        \vb{u} = {\sum_{\vb{b} \in \basis}}' \lambda_{\vb{b}} \vb{b} \qand \vb{v} = {\sum_{\vb{b} \in \basis}}' \mu_{\vb{b}} \vb{b}.
    \]

    Then
    \begin{align*}
        F\pqty{\vb{u} + \lambda \vb{v}} &= F\pqty{{\sum_{\vb{b} \in \basis}}' \lambda_{\vb{b}} \vb{b} + {\sum_{\vb{b} \in \basis}}' \lambda \mu_{\vb{b}} \vb{b}} \\
        &= F\pqty{{\sum_{\vb{b} \in \basis}}' \pqty{\lambda_{\vb{b}} + \lambda \mu_{\vb{b}}} \vb{b}} \\
        &= {\sum_{\vb{b} \in \basis}}' \pqty{\lambda_{\vb{b}} + \lambda \mu_{\vb{b}}} f\pqty{\vb{b}} \\
        &= {\sum_{\vb{b} \in \basis}}' \lambda_{\vb{b}} f\pqty{\vb{b}} + {\sum_{\vb{b} \in \basis}}' \lambda \mu_{\vb{b}} f\pqty{\vb{b}} \\
        &= F\pqty{\vb{u}} + \lambda F\pqty{\vb{v}}.
    \end{align*}
\end{proof}
